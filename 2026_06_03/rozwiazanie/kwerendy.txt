1. SELECT nazwa, styl, cena FROM meble WHERE cena <= 1000 AND czyNowy = 1 ORDER BY cena ASC;
2. CREATE VIEW komody AS SELECT idMeble, nazwa, kategoria, cena FROM meble WHERE kategoria = 3;
3. SELECT ROUND(AVG(cena)) AS "Średnia cena komód", MIN(cena) AS "Minimalna", MAX(cena) AS "Maksymalna" FROM komody;
